GLOBAL INT AXIS9
INT RE1
GLOBAL REAL ZD1
RE1=0
AXIS9=1
VEL(AXIS9)=100!Velocity,mm/s
DISABLE(AXIS9)
ASSIGNPEG AXIS9,0b0001,0b0001
ENABLE(AXIS9)
SET RPOS(AXIS9)=0
CYCLE:
PEG_I 0,0.5,5,50,1000
!5 STARTPOSITION 100 ENDPOSITION 
TILL AST(0).#PEGREADY
PTP/E AXIS9,1000
WAIT 1000
PTP/E AXIS9,0
WAIT 1000
IF RE1
GOTO CYCLE
END
STOP
